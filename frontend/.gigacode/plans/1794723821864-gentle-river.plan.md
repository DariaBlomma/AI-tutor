# План: Student Dashboard (root route `/`)

> **Code Style Conventions проекта** (изучены через анализ существующего кода):
> - **SCSS:** `@use '@/shared/styles/variables' as *;` → `$p`, `$c1`, `$c3`, `$err`, `$ok`, `$radius`, `$breakpoint-desktop` (НЕ `var(--xxx)`)
> - **Hover:** `&:hover { filter: brightness(1.1); }` (НЕ `transform`, НЕ `border-color`)
> - **Active:** НЕ используем `transform: scale`. Только цвет/filter.
> - **Media:** `@media (min-width: $breakpoint-desktop)` (НЕ хардкод `1024px`)
> - **Импорты:** React → External libs → Local `@/` → Local styles
> - **Экспорт:** Named exports ONLY (no default)
> - **Props types:** inline interface в `.tsx` файле
> - **CSS class composition:** template literals `${styles.a} ${styles.b}`
> - **Hooks:** `use{Domain}{Purpose}`, return object with named fields
> - **Services:** Class with static async methods
> - **BEM:** NOT used — simple descriptive names
> - **Именование модуля Student:** все компоненты с префиксом `Student` (StudentDashboardHeader, StudentStatCards, StudentAssignmentItem, StudentStatusBadge, StudentAssignmentList)
> - **Компонент = только JSX** — ни логики, ни хелперов внутри
> - **UI хелперы** (pluralizeAssignments, formatDeadline, toStatusType) → `src/student/utils/`
> - **Логика** (fetch, state, side effects) → хуки в `src/student/hooks/`
> - **Helper functions** внизу файла, после `export function`
> - **Sub-components:** переиспользуемые → отдельные файлы, вспомогательные → sibling functions

## Контекст

**Бэкенд:** `http://localhost:8080/api`

### API endpoint'ы (студент)

| Endpoint | Method | Response | Описание |
|---|---|---|---|
| `/student/work` | GET | `Page<StudentWorkResponse>` | Список всех заданий |
| `/student/me/progress` | GET | `StudentProgressOverviewResponse` | Прогресс по активным |
| `/user/me` | GET | `UserProfile` | Профиль (имя + stats) |

**Отдельного `/student/dashboard` НЕ СУЩЕСТВУЕТ.** Данные собираются из endpoint'ов.

### Ключевые DTO бэкенда

**StudentWorkResponse** (`GET /student/work`):
```java
Long id; Long taskId; String taskPublicId; String taskName;
WorkStatus status;  // ASSIGNED | IN_PROGRESS | COMPLETED | CANCELLED | EXPIRED
Instant deadline; Instant assignedAt; Instant startedAt; Instant completedAt;
```

**StudentStatsDto** (вложен в `GET /user/me`):
```java
Integer completedCount; Long avgTimeSeconds; Long errorsCount;
```

### Что нужно vs что есть

| Требование | Реальность | Решение |
|---|---|---|
| Статистика за **текущий месяц** | `StudentStatsDto` — все время | Считаем на фронте из `/student/work` |
| Статусы: Новое/В процессе/Просрочено | Бэк: `ASSIGNED/IN_PROGRESS/EXPIRED` | `ASSIGNED` → "Новое", `IN_PROGRESS` → "В процессе", `EXPIRED` → "Просрочено" |
| Список ДЗ | `GET /student/work` — уже есть | Используем |
| Имя ученика | `GET /user/me` — уже есть | Используем |

```ts
import { api } from '@/core/api/AxiosInstance';
import type { WorkStatus } from '@/student/types/work';

const STUDENT_PATH = '/student';

export interface StudentWorkResponse {
    id: number;
    taskId: number;
    taskPublicId: string;
    taskName: string;
    status: WorkStatus;
    deadline: string;
    assignedAt: string;
    startedAt: string;
    completedAt: string;
}

export interface StudentWorkPage {
    content: StudentWorkResponse[];
    totalElements: number;
    totalPages: number;
    number: number;
    size: number;
}

export class StudentDashboardService {
    static async getWorks(page = 0, size = 100): Promise<StudentWorkPage> {
        const response = await api.get<StudentWorkPage>(`${STUDENT_PATH}/work`, {
            params: { page, size },
        });
        return response.data;
    }

    static async getProgress() {
        const response = await api.get(`${STUDENT_PATH}/me/progress`);
        return response.data;
    }
}
```

## Шаг 3: Утилиты

**Файл:** `../../src/student/utils/dashboardUtils.ts` (изменить — добавить)

```ts
import type { StudentWorkResponse } from '@/student/services/StudentDashboardService';
import type { MonthlyStats, DashboardAssignment } from '@/student/types/models/dashboard';
import { toDashboardStatus } from '@/student/types/work';

export function getMonthBounds(): { start: number; end: number } {
    const now = new Date();
    const start = new Date(now.getFullYear(), now.getMonth(), 1).getTime();
    const end = new Date(now.getFullYear(), now.getMonth() + 1, 0, 23, 59, 59, 999).getTime();
    return { start, end };
}

export function mapToDashboardAssignment(work: StudentWorkResponse): DashboardAssignment {
    return {
        id: work.id,
        taskPublicId: work.taskPublicId,
        taskName: work.taskName,
        status: toDashboardStatus(work.status),
        rawStatus: work.status,
        deadline: work.deadline,
    };
}

export function computeMonthlyStats(works: DashboardAssignment[]): MonthlyStats {
    const { start, end } = getMonthBounds();
    const monthWorks = works.filter(w => {
        const ts = new Date(w.deadline).getTime();
        return ts >= start && ts <= end;
    });
    return {
        completedCount: monthWorks.filter(w => w.rawStatus === 'COMPLETED').length,
        inProgressCount: monthWorks.filter(w => w.rawStatus === 'ASSIGNED' || w.rawStatus === 'IN_PROGRESS').length,
        overdueCount: monthWorks.filter(w => w.rawStatus === 'EXPIRED').length,
    };
}
```

```ts
export function pluralizeAssignments(count: number): string {
    const abs = Math.abs(count) % 100;
    const lastDigit = abs % 10;
    if (abs > 10 && abs < 20) return 'заданий';
    if (lastDigit > 1 && lastDigit < 5) return 'задания';
    if (lastDigit === 1) return 'задание';
    return 'заданий';
}

export function formatDeadline(isoString: string): string {
    const date = new Date(isoString);
    const months = ['янв', 'фев', 'мар', 'апр', 'мая', 'июн', 'июл', 'авг', 'сен', 'окт', 'ноя', 'дек'];
    const day = date.getDate();
    const month = months[date.getMonth()];
    return `${day} ${month}`;
}

export function toDashboardStatusLabel(status: string): 'new' | 'in-progress' | 'overdue' {
    return status.toLowerCase().replace(/ /g, '-') as 'new' | 'in-progress' | 'overdue';
}
```

## Шаг 4: Хук

**Файл:** `src/student/hooks/useStudentDashboard.ts` (новый)

```ts
import { useQuery } from '@tanstack/react-query';
import { StudentDashboardService } from '@/student/services/StudentDashboardService';
import { extractErrorMessage } from '@/core/api/utils/errors';
import { mapToDashboardAssignment, computeMonthlyStats } from '@/student/lib/dashboardUtils';
import type { MonthlyStats, DashboardAssignment } from '@/student/types/models/dashboard';

export interface UseStudentDashboardResult {
    stats: MonthlyStats | null;
    assignments: DashboardAssignment[];
    isLoading: boolean;
    isError: boolean;
    error: string | null;
    refetch: () => void;
}

export function useStudentDashboard(): UseStudentDashboardResult {
    const query = useQuery({
        queryKey: ['student', 'dashboard'],
        queryFn: () => StudentDashboardService.getWorks(0, 100),
    });

    if (query.isLoading) {
        return { stats: null, assignments: [], isLoading: true, isError: false, error: null, refetch: query.refetch };
    }

    if (query.isError) {
        return {
            stats: null, assignments: [], isLoading: false, isError: true,
            error: extractErrorMessage(query.error), refetch: query.refetch,
        };
    }

    const works = query.data?.content ?? [];
    const assignments = works.map(mapToDashboardAssignment);
    const stats = computeMonthlyStats(assignments);

    return { stats, assignments, isLoading: false, isError: false, error: null, refetch: query.refetch };
}
```

```tsx
            {newAssignmentsCount > 0 && (
                <p>У вас {newAssignmentsCount} {assignmentsText}</p>
            )}
        </div>
    );
}
```

## Шаг 6: Обновить StudentHomeView

**Файл:** `src/student/ui/views/StudentHomeView/StudentHomeView.tsx` (изменить)

```tsx
import { Loader } from '@/shared/ui-kit/Loader/Loader';
import { useStudentDashboard } from '@/student/hooks/useStudentDashboard';
import { StudentDashboardHeader } from '@/student/ui/components/DashboardHeader/StudentDashboardHeader';

import { StudentAssignmentList } from '@/student/ui/components/AssignmentList/StudentAssignmentList';
import styles from './StudentHomeView.module.scss';

export function StudentHomeView() {
    const { stats, assignments, isLoading, isError, error } = useStudentDashboard();

    const newCount = assignments.filter(a => a.status === 'NEW').length;

    return (
        <div className={styles.page}>
            {isLoading && !stats ? (
                <Loader active />
            ) : isError ? (
                <div className={styles.error}>
                    <p>Не удалось загрузить данные</p>
                    {error && <p className={styles.errorDetail}>{error}</p>}
                </div>

                <>
                    <StudentDashboardHeader newAssignmentsCount={newCount} />
                    {stats && <StudentStatCards stats={stats} />}
                    <StudentAssignmentList assignments={assignments} isLoading={isLoading} />
                </>
            )}
        </div>
    );
}
```
