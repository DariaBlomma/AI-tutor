## Шаг 1: Миграция `useTaskDetail` на TanStack Query

**Файл:** `src/tasks/hooks/useTaskDetail.ts`

**Текущее состояние:** ручной `useState` + `useEffect` паттерн.

**Изменения:**
1. Заменить на `useQuery` из `@tanstack/react-query`
2. Убрать параметр `fetcher` (не используется внешними вызывающими)
3. Сохранить интерфейс `UseTaskDetailResult` для обратной совместимости
4. `staleTime: 0` — без кэширования (как сейчас)
5. `retry: false` — без повторных попыток

**Код:**
```ts
import { useQuery } from '@tanstack/react-query';
import { TasksBankService } from '@/tasks/services/TasksBankService';
import { extractErrorMessage } from '@/core/api/utils/errors';
import type { TaskResponseDto } from '@/tasks/types';

export interface UseTaskDetailResult {
    task: TaskResponseDto | null;
    isLoading: boolean;
    isError: boolean;
    error: string | null;
    reload: () => void;
}

export function useTaskDetail(publicId: string): UseTaskDetailResult {
    const query = useQuery({
        queryKey: ['task', publicId],
        queryFn: () => TasksBankService.getTaskByPublicId(publicId),
        staleTime: 0,
        retry: false,
    });

    return {
        task: query.data ?? null,
        isLoading: query.isLoading,
        isError: query.isError,
        error: query.error ? extractErrorMessage(query.error) : null,
        reload: query.refetch,
    };
}
```

import type { TaskItemResponseDto } from '@/tasks/types';
import { parseMatchingContent } from './parseTaskContent';

/**
 * Человекочитаемый правильный ответ пункта задания.
 * У MATCHING нет correctAnswer на бэке — собираем из content.
 */
export function formatCorrectAnswer(item: TaskItemResponseDto): string {
    if (item.correctAnswer) return item.correctAnswer;
    if (item.itemType === 'MATCHING') {
        const pair = parseMatchingContent(item.content);
        if (pair?.left_label) return `${pair.left_label} → ${pair.right_label ?? ''}`;
    }
    return '—';
}

## Шаг 3: Создание хука `useTaskDetailCourses`

**Новый файл:** `src/tasks/hooks/useTaskDetailCourses.ts`

**Содержимое:**
```ts
import { useCallback, useEffect, useState } from 'react';
import { TasksBankService } from '@/tasks/services/TasksBankService';

export interface UseTaskDetailCoursesResult {
    selectedCourseIds: string[];
    actionError: string | null;
    handleCourseChange: (values: string[]) => Promise<void>;
}

/**
 * Управление выбором курсов для задания на странице просмотра.
 * Инициализация из task.courseIds, сохранение через API.
 */
export function useTaskDetailCourses(
    taskId: number | undefined,
    initialCourseIds: number[] | undefined,
): UseTaskDetailCoursesResult {
    const [selectedCourseIds, setSelectedCourseIds] = useState<string[]>([]);
    const [actionError, setActionError] = useState<string | null>(null);

    useEffect(() => {
        if (initialCourseIds) {
            setSelectedCourseIds(initialCourseIds.map((id) => String(id)));
        } else {
            setSelectedCourseIds([]);
        }
    }, [initialCourseIds]);

    const handleCourseChange = useCallback(
        async (values: string[]) => {
            setSelectedCourseIds(values);
            setActionError(null);
            if (!taskId) return;

            const courseIdNums = values.map(Number);
            try {
                await TasksBankService.updateTaskCourseIds(taskId, courseIdNums);
            } catch (e) {
                setActionError(e instanceof Error ? e.message : 'Не удалось обновить курсы.');
            }
        },
        [taskId],
    );

    return { selectedCourseIds, actionError, handleCourseChange };
}
```

**TaskDetailItem.tsx:**
```tsx
import type { TaskItemResponseDto } from '@/tasks/types';
import {
    parseWordFormationContent,
    parseGapFillContent,
    parseMatchingContent,
    parseExplanation,
} from '@/tasks/lib/parseTaskContent';
import { formatCorrectAnswer } from '@/tasks/lib/formatCorrectAnswer';
import styles from './TaskDetailItem.module.scss';

interface TaskDetailItemProps {
    item: TaskItemResponseDto;
    showAnswers: boolean;
}

export function TaskDetailItem({ item, showAnswers }: TaskDetailItemProps) {
    const word = parseWordFormationContent(item.content);
    const gap = parseGapFillContent(item.content);
    const pair = parseMatchingContent(item.content);
    const explanation = parseExplanation(item.content);

    return (
        <li className={styles.item}>
            <div className={styles.itemHeader}>Пункт {item.position + 1}</div>

            {item.prompt && <p className={styles.itemPrompt}>{item.prompt}</p>}

            {item.itemType === 'WORD_FORMATION' && word?.base && (
                <p className={styles.itemMeta}>
                    База: <strong>{word.base}</strong>
                </p>
            )}
            {item.itemType === 'GAP_FILL' && gap?.options?.length ? (
                <p className={styles.itemMeta}>Опции: {gap.options.join(', ')}</p>
            ) : null}
            {item.itemType === 'MATCHING' && pair?.left_label && (
                <p className={styles.itemMeta}>
                    {pair.left_label} → {pair.right_label ?? ''}
                </p>
            )}

            {showAnswers && (
                <p className={styles.itemAnswer}>
                    Ответ: <strong>{formatCorrectAnswer(item)}</strong>
                </p>
            )}
            {explanation && <p className={styles.itemExplanation}>Пояснение: {explanation}</p>}
        </li>
    );
}
```

    showAnswers: boolean;
